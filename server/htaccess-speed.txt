# Sound Creations Rwanda - speed rules for /public_html/.htaccess (LiteSpeed / Apache)
# Paste this block at the TOP of public_html/.htaccess (above "# BEGIN WordPress").
# Keep a copy of the original .htaccess first.

# --- Security: hide version/info files and block code execution in uploads ---
<FilesMatch "^(readme\.html|license\.txt|wp-config-sample\.php|wp-config\.php|xmlrpc\.php|debug\.log|\.env|composer\.(json|lock))$">
  Require all denied
</FilesMatch>
<IfModule mod_rewrite.c>
  RewriteEngine On
  # No PHP may ever run from uploads (stops uploaded web shells)
  RewriteRule ^wp-content/uploads/.*\.(php[0-9]?|phtml|phar)$ - [F,L,NC]
  # Block common exploit probes
  RewriteRule (^|/)\.(git|svn|env|htpasswd) - [F,L,NC]
</IfModule>
Options -Indexes

# --- Compression (text files) ---
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/xml application/javascript application/json application/xml image/svg+xml font/woff2
</IfModule>

# --- Browser caching: static files for 1 year, HTML not cached by the browser ---
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/webp "access plus 1 year"
  ExpiresByType image/jpeg "access plus 1 year"
  ExpiresByType image/png "access plus 1 year"
  ExpiresByType image/svg+xml "access plus 1 year"
  ExpiresByType image/x-icon "access plus 1 year"
  ExpiresByType video/mp4 "access plus 1 year"
  ExpiresByType font/woff2 "access plus 1 year"
  ExpiresByType text/css "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 year"
  ExpiresByType text/html "access plus 0 seconds"
</IfModule>
<IfModule mod_headers.c>
  <FilesMatch "\.(webp|jpe?g|png|svg|ico|mp4|woff2|css|js)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

# --- Page cache (biggest win for a 1-3 s load) ---

# Image Optimization > WebP replacement ON; Object cache (Redis/Memcached) if the host offers it.
